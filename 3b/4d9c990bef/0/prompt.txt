```go
type ExecRunner struct{}

func (ExecRunner) Run(ctx context.Context, dir, name string, args ...string) ([]byte, []byte, error) {
    cmd := exec.CommandContext(ctx, name, args...)
    if dir != "" {
        cmd.Dir = dir
    }

    var stdout bytes.Buffer
    var stderr bytes.Buffer
    cmd.Stdout = &amp;stdout
    cmd.Stderr = &amp;stderr

    err := cmd.Run()
    if err != nil {
        if stderr.Len() &gt; 0 {
            return stdout.Bytes(), stderr.Bytes(), fmt.Errorf("%s %v: %w: %s", name, args, err, stderr.String())
        }
        return stdout.Bytes(), stderr.Bytes(), fmt.Errorf("%s %v: %w", name, args, err)
    }
    return stdout.Bytes(), stderr.Bytes(), nil
}
```

**Key Features:**
- Uses `exec.CommandContext` for timeout support
- **Full stdout capture**: `bytes.Buffer` attached to `cmd.Stdout`, entire output returned as `[]byte` via `stdout.Bytes()`
- **Full stderr capture**: Same pattern for stderr
- No streaming—complete output buffered in memory

```go
    defer cancel()
    stdout, stderr, err := runner.Run(runCtx, repoDir, semBinary, args...)

        return nil, fmt.Errorf("semantic provider snapshot timed out after %s", semanticSnapshotTimeout)
```

```go
func populateSemanticSQLite(db *sql.DB, repoDir, generationID string, raw []byte, header semanticHeader, counts semanticCounts, now time.Time, generationDir string) (semanticBuildMetrics, error) {
    tx, err := db.Begin()
    if err != nil {
        return semanticBuildMetrics{}, err
    }
    // ... commit/error handlers ...
    
    // Insert metadata
    if _, err := tx.Exec(`INSERT INTO meta(key, value) VALUES (?, ?), (?, ?), (?, ?), (?, ?)`,
        "schema_version", header.SchemaVersion,
        "provider", header.Provider,
        "provider_version", header.ProviderVersion,
        "generation_id", generationID,
    ); err != nil {
        return commit(err)
    }
    
    // Insert warnings and partial failures
    for _, warning := range header.Warnings {
        if err := insertSemanticWarning(tx, warning, "warning"); err != nil {
            return commit(err)
        }
    }
    for _, failure := range header.PartialFailures {
        if err := insertSemanticWarning(tx, failure, "partial_failure"); err != nil {
            return commit(err)
        }
    }
    
    // Parse NDJSON records and ingest
    files := make(map[string]semanticRecord)
    scanner := newSemanticScanner(bytes.NewReader(raw))
    if !scanner.Scan() {
        return commit(errors.New("semantic snapshot missing header"))
    }
    for scanner.Scan() {
        text := bytes.TrimSpace(scanner.Bytes())
        if len(text) == 0 {
            continue
        }
        var record semanticRecord
        if err := json.Unmarshal(text, &amp;record); err != nil {
            return commit(err)
        }
        switch record.RecordType {
        case "file":
            path := record.semanticPath()
            if path != "" {
                path, err = validateSemanticProviderPath(path)
                if err != nil {
                    return commit(err)
                }
                record.setSemanticPath(path)
                files[path] = record
            }
        case "symbol":
            path := record.semanticPath()
            if path != "" {
                path, err = validateSemanticProviderPath(path)
                if err != nil {
                    return commit(err)
                }
                record.setSemanticPath(path)
                if existing, ok := files[path]; ok {
                    if strings.TrimSpace(existing.Language) == "" &amp;&amp; strings.TrimSpace(record.Language) != "" {
                        existing.Language = record.Language
                        files[path] = existing
                    }
                } else {
                    files[path] = semanticRecord{RecordType: "file", FilePath: path}
                }
            }
            if err := insertSemanticSymbol(tx, record); err != nil {
                return commit(err)
            }
        case "relation":
            if err := insertSemanticRelation(tx, record); err != nil {
                return commit(err)
            }
        }
    }
    // ... build parse cache ...
    // ... metrics ...
}
```

## Summary: Invocation Chain

1. **Provider invocation** → `entire sem snapshot --repo &lt;repo&gt; --format ndjson --no-network [--ignore-file ...] [--worktree]`
   - Binary: configurable via `--sem-binary` flag (default: `"entire"`)
   - Timeout: 2 minutes (`context.WithTimeout`)
   - Output: NDJSON to stdout, fully buffered in `[]byte`

2. **Output capture** → `exec.CommandContext` with `bytes.Buffer` on stdout/stderr
   - No streaming; entire snapshot held in memory

3. **Filtering** → `filterSemanticSnapshot(raw []byte, ignore brainIgnore, repoDir string)` (two-pass scan)
   - Pass 1: Build ignored symbol ID map
   - Pass 2: Filter records, count symbols/relations/files, sanitize warnings
   - Uses `bufio.Scanner` with 64 MB max record size

4. **Manifest building** → Extract header metadata, fill provider/commit/tree defaults, record symbol/relation counts, warnings, capabilities, no-egress verification status

5. **Snapshot write** → NDJSON artifact to `.entire/semantic/snapshots/&lt;commit&gt;-&lt;sha256[:8]&gt;/snapshot.ndjson`

6. **SQLite ingestion** → `populateSemanticSQLite` (third pass through raw NDJSON):
   - Header → meta table
   - Warnings/partial_failures → warnings table (with source tracking)
   - Symbols → symbols table + symbol_fts FTS index
   - Relations → relations + reverse_relations tables
   - Files → files table (from symbols + symbol records)
   - Parse cache → JSON artifacts per unique content hash
